        let tasks = [];

        let currentFilter = 'all';

        let activeTaskId = null;

        let timerInterval = null;

        let timeLeft = 25 * 60;

        let isTimerRunning = false;

 

        // Initialization

        document.addEventListener('DOMContentLoaded', () => {

            initTheme();

            const saved = localStorage.getItem('execuflow_pro_tasks');

            if (saved) {

                generateSmartDay(true);

        function saveTasks() {

            localStorage.setItem('execuflow_pro_tasks', JSON.stringify(tasks));

            updateStats();

        }

 

        // Generate full balanced day

        function generateSmartDay(silent = false) {

            const smartSelection = [

                PROTOCOLS.deepwork[0],

                PROTOCOLS.deepwork[1],

                PROTOCOLS.salud[0],

                PROTOCOLS.ejecutivo[0],

                PROTOCOLS.logistica[1],

                PROTOCOLS.ejecutivo[2]

            ];

 

            smartSelection.forEach(item => {

                const exists = tasks.some(t => t.title === item.title && !t.completed);

                if (!exists) {

                    tasks.push({

                        id: Date.now() + Math.random(),

                        title: item.title,

                        category: item.category,

                        priority: item.priority,

                        duration: item.duration,

                        instructions: item.instructions,

                        steps: item.steps.map(s => ({ text: s, done: false })),

                        completed: false

                    });

                }

            });

 

            saveTasks();

            renderTasks();

            if (!silent) showToast("✨ Día Inteligente generado con instrucciones paso a paso");

        }

 

        function loadProtocol(key) {

            const list = PROTOCOLS[key];

            if (!list) return;

            list.forEach(item => {

                tasks.unshift({

                    id: Date.now() + Math.random(),

                    title: item.title,

                    category: item.category,

                    priority: item.priority,

                    duration: item.duration,

                    instructions: item.instructions,

                    steps: item.steps.map(s => ({ text: s, done: false })),

                    completed: false

                });

            });

            saveTasks();

            renderTasks();

            showToast("📦 Protocolo profesional cargado en tu lista");

        }

 

        function autoFillInstructions() {

            const cat = document.getElementById('task-category').value;

            const title = document.getElementById('task-title').value.trim();

            const tpl = DEFAULT_TEMPLATES[cat] || DEFAULT_TEMPLATES["Trabajo"];

 

            document.getElementById('task-instructions').value = title

                ? `Para «${title}»: ${tpl.instructions}`

                : tpl.instructions;

            document.getElementById('task-steps').value = tpl.steps.join(', ');

            showToast("🪄 Instrucciones y pasos generados automáticamente");

        }

 

        function addCustomTask(e) {

            e.preventDefault();

            const title = document.getElementById('task-title').value.trim();

            const category = document.getElementById('task-category').value;

            const priority = document.getElementById('task-priority').value;

            const duration = parseInt(document.getElementById('task-duration').value, 10);

            let instructions = document.getElementById('task-instructions').value.trim();

            let stepsRaw = document.getElementById('task-steps').value.trim();

 

            const tpl = DEFAULT_TEMPLATES[category];

            if (!instructions) instructions = tpl.instructions;

            const stepsArray = stepsRaw

                ? stepsRaw.split(',').map(s => s.trim()).filter(Boolean)

                : tpl.steps;

 

            const newTask = {

                id: Date.now(),

                title,

                category,

                priority,

                duration,

                instructions,

                steps: stepsArray.map(s => ({ text: s, done: false })),

                completed: false

            };

 

            tasks.unshift(newTask);

            saveTasks();

            renderTasks();

            e.target.reset();

            showToast("✅ Actividad agregada con su guía de ejecución");

        }

 

        // Autopilot / Focus Panel

        function startAutopilot(specificId = null) {

            const priorityOrder = { 'Alta': 1, 'Media': 2, 'Baja': 3 };

            let targetTask = null;

 

            if (specificId) {

                targetTask = tasks.find(t => t.id === specificId);

            } else {

                const pending = tasks.filter(t => !t.completed).sort((a, b) => priorityOrder[a.priority] - priorityOrder[b.priority]);

                targetTask = pending[0];

            }

 

            if (!targetTask) {

                showToast("🎉 ¡No tienes actividades pendientes! Genera un Día Inteligente.");

                return;

            }

 

            activeTaskId = targetTask.id;

            document.getElementById('focus-category').textContent = `${targetTask.category} • PRIORIDAD ${targetTask.priority.toUpperCase()}`;

            document.getElementById('focus-title').textContent = targetTask.title;

            document.getElementById('focus-desc').textContent = targetTask.instructions;

            document.getElementById('complete-focus-btn').classList.remove('hidden');

 

            renderFocusSteps(targetTask);

 

            // Configure timer

            pauseTimer();

            timeLeft = (targetTask.duration || 25) * 60;

            updateTimerDisplay();

            renderTasks();

            window.scrollTo({ top: 0, behavior: 'smooth' });

        }

 

        function renderFocusSteps(task) {

            const container = document.getElementById('focus-steps-container');

            const listEl = document.getElementById('focus-steps-list');

            if (!task.steps || task.steps.length === 0) {

                container.classList.add('hidden');

                return;

            }

            container.classList.remove('hidden');

            listEl.innerHTML = task.steps.map((step, idx) => `

                <label class="flex items-start gap-2.5 cursor-pointer p-1.5 rounded-lg hover:bg-white/10 transition">

                    <input type="checkbox" ${step.done ? 'checked' : ''} onchange="toggleSubstep(${task.id}, ${idx})" class="mt-1 rounded text-emerald-500 focus:ring-0">

                    <span class="${step.done ? 'line-through text-indigo-200/60' : 'text-white'}">${step.text}</span>

                </label>

            `).join('');

        }

 

        function toggleSubstep(taskId, stepIdx) {

            const task = tasks.find(t => t.id === taskId);

            if (!task) return;

            task.steps[stepIdx].done = !task.steps[stepIdx].done;

            saveTasks();

            if (activeTaskId === taskId) renderFocusSteps(task);

            renderTasks();

        }

 

        function completeFocusedTask() {

            if (!activeTaskId) return;

            const task = tasks.find(t => t.id === activeTaskId);

            if (task) {

                task.completed = true;

                task.steps.forEach(s => s.done = true);

                saveTasks();

                showToast("🏆 ¡Excelente! Actividad completada.");

            }

            const remaining = tasks.filter(t => !t.completed);

            if (remaining.length > 0) {

                startAutopilot();

            } else {

                activeTaskId = null;

                pauseTimer();

                document.getElementById('focus-title').textContent = "🎉 ¡Día Completado con Éxito!";

                document.getElementById('focus-desc').textContent = "Has terminado todas las actividades programadas. Descansa o carga un nuevo protocolo.";

                document.getElementById('focus-steps-container').classList.add('hidden');

                document.getElementById('complete-focus-btn').classList.add('hidden');

                renderTasks();

            }

        }

 

        // Timer Functions

        function updateTimerDisplay() {

            const m = String(Math.floor(timeLeft / 60)).padStart(2, '0');

            const s = String(timeLeft % 60).padStart(2, '0');

            document.getElementById('timer-display').textContent = `${m}:${s}`;

        }

 

        function toggleTimer() {

            if (isTimerRunning) {

                pauseTimer();

            } else {

                isTimerRunning = true;

                document.getElementById('timer-btn').innerHTML = `<i class="fa-solid fa-pause mr-1"></i> Pausar`;

                timerInterval = setInterval(() => {

                    if (timeLeft > 0) {

                        timeLeft--;

                        updateTimerDisplay();

                    } else {

                        pauseTimer();

                        playBeep();

                        showToast("⏰ ¡Tiempo del bloque terminado!");

                    }

                }, 1000);

            }

        }

 

        function pauseTimer() {

            isTimerRunning = false;

            clearInterval(timerInterval);

            const btn = document.getElementById('timer-btn');

            if (btn) btn.innerHTML = `<i class="fa-solid fa-play mr-1"></i> Iniciar`;

        }

 

        function resetTimer() {

            pauseTimer();

            const task = tasks.find(t => t.id === activeTaskId);

            timeLeft = ((task && task.duration) ? task.duration : 25) * 60;

            updateTimerDisplay();

        }

 

        function playBeep() {

            try {

                const ctx = new (window.AudioContext || window.webkitAudioContext)();

                const osc = ctx.createOscillator();

                const gain = ctx.createGain();

                osc.type = 'sine';

                osc.frequency.setValueAtTime(660, ctx.currentTime);

                gain.gain.setValueAtTime(0.15, ctx.currentTime);

                osc.connect(gain);

                gain.connect(ctx.destination);

                osc.start();

                osc.stop(ctx.currentTime + 0.4);

            } catch (e) {}

        }

 

        // Task Operations & Rendering

        function toggleTaskStatus(id) {

            const task = tasks.find(t => t.id === id);

            if (!task) return;

            task.completed = !task.completed;

            saveTasks();

            renderTasks();

        }

 

        function deleteTask(id) {

            tasks = tasks.filter(t => t.id !== id);

            if (activeTaskId === id) activeTaskId = null;

            saveTasks();

            renderTasks();

        }

 

        function clearCompleted() {

            tasks = tasks.filter(t => !t.completed);

            saveTasks();

            renderTasks();

            showToast("🧹 Actividades completadas eliminadas");

        }

 

        function setFilter(f) {

            currentFilter = f;

            ['all', 'pending', 'completed'].forEach(k => {

                const btn = document.getElementById(`filter-${k}`);

                if (k === f) {

                    btn.className = "flex-1 sm:flex-initial px-3 py-1.5 rounded-lg text-xs font-semibold bg-brand-600 text-white transition";

                } else {

                    btn.className = "flex-1 sm:flex-initial px-3 py-1.5 rounded-lg text-xs font-semibold text-slate-600 dark:text-slate-400 hover:text-slate-900 transition";

                }

            });

            renderTasks();

        }

 

        function updateStats() {

            const total = tasks.length;

            const completed = tasks.filter(t => t.completed).length;

            const pending = total - completed;

            const totalMinutes = tasks.filter(t => !t.completed).reduce((acc, t) => acc + (t.duration || 25), 0);

            const pct = total === 0 ? 0 : Math.round((completed / total) * 100);

 

            document.getElementById('stat-pending').textContent = pending;

            document.getElementById('stat-completed').textContent = completed;

            document.getElementById('stat-time').textContent = `${totalMinutes}m`;

            document.getElementById('progress-text').textContent = `${pct}%`;

            document.getElementById('progress-bar').style.width = `${pct}%`;

        }

 

        function renderTasks() {

            updateStats();

            const query = document.getElementById('search-input').value.toLowerCase();

            const container = document.getElementById('tasks-list');

 

            const filtered = tasks.filter(t => {

                if (currentFilter === 'pending' && t.completed) return false;

                if (currentFilter === 'completed' && !t.completed) return false;

                if (query && !t.title.toLowerCase().includes(query) && !t.category.toLowerCase().includes(query)) return false;

                return true;

            });

 

            if (filtered.length === 0) {

                container.innerHTML = `

                    <div class="bg-white dark:bg-slate-800 rounded-2xl p-8 text-center border border-dashed border-slate-300 dark:border-slate-700">

                        <i class="fa-solid fa-clipboard-check text-3xl text-slate-400 mb-2"></i>

                        <p class="font-semibold text-sm">No hay actividades en esta vista</p>

                        <p class="text-xs text-slate-500 mt-1">Usa el botón «Generar Día Inteligente» arriba para armar tu rutina.</p>

                    </div>

                `;

                return;

            }

 

            const priorityBadge = {

                'Alta': 'bg-rose-100 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300',

                'Media': 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300',

                'Baja': 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300'

            };

 

            container.innerHTML = filtered.map(task => {

                const isFocused = task.id === activeTaskId;

                const stepsDone = task.steps ? task.steps.filter(s => s.done).length : 0;

                const totalSteps = task.steps ? task.steps.length : 0;

 

                return `

                <div class="bg-white dark:bg-slate-800 rounded-2xl p-4 border ${isFocused ? 'border-2 border-brand-500 ring-2 ring-brand-500/20' : 'border-slate-200 dark:border-slate-700'} shadow-sm transition">

                    <div class="flex items-start justify-between gap-3">

                        <div class="flex items-start gap-3 flex-1">

                            <input type="checkbox" ${task.completed ? 'checked' : ''} onchange="toggleTaskStatus(${task.id})" class="mt-1 w-4 h-4 rounded text-brand-600 focus:ring-brand-500 cursor-pointer">

                            <div class="flex-1">

                                <div class="flex flex-wrap items-center gap-2 mb-1">

                                    <span class="text-[11px] font-semibold px-2 py-0.5 rounded-md bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-300">${task.category}</span>

                                    <span class="text-[11px] font-semibold px-2 py-0.5 rounded-md ${priorityBadge[task.priority] || ''}">${task.priority}</span>

                                    <span class="text-[11px] text-slate-400"><i class="fa-regular fa-clock mr-1"></i>${task.duration} min</span>

                                    ${totalSteps > 0 ? `<span class="text-[11px] text-brand-600 dark:text-brand-400 font-medium"><i class="fa-solid fa-list-check mr-1"></i>${stepsDone}/${totalSteps} pasos</span>` : ''}

                                </div>

                                <h4 class="font-bold text-sm ${task.completed ? 'line-through text-slate-400' : 'text-slate-800 dark:text-slate-100'}">${task.title}</h4>

                                <p class="text-xs text-slate-500 dark:text-slate-400 mt-1 leading-relaxed">${task.instructions}</p>

 

                                ${task.steps && task.steps.length > 0 ? `

                                <div class="mt-3 pt-2.5 border-t border-slate-100 dark:border-slate-700/60 space-y-1.5">

                                    ${task.steps.map((st, sIdx) => `

                                        <label class="flex items-center gap-2 text-xs cursor-pointer text-slate-600 dark:text-slate-300 hover:text-brand-600">

                                            <input type="checkbox" ${st.done ? 'checked' : ''} onchange="toggleSubstep(${task.id}, ${sIdx})" class="rounded text-brand-600 focus:ring-0 w-3.5 h-3.5">

                                            <span class="${st.done ? 'line-through text-slate-400' : ''}">${st.text}</span>

                                        </label>

                                    `).join('')}

                                </div>

                                ` : ''}

                            </div>

                        </div>

 

                        <div class="flex items-center gap-1.5">

                            ${!task.completed ? `

                            <button onclick="startAutopilot(${task.id})" class="px-2.5 py-1.5 rounded-lg bg-brand-50 dark:bg-brand-900/30 text-brand-600 dark:text-brand-400 hover:bg-brand-600 hover:text-white text-xs font-semibold transition flex items-center gap-1" title="Enfocarme en esta actividad">

                                <i class="fa-solid fa-crosshairs"></i> Guiarme

                            </button>

                            ` : ''}

                            <button onclick="deleteTask(${task.id})" class="p-1.5 rounded-lg text-slate-400 hover:text-rose-500 hover:bg-rose-50 dark:hover:bg-slate-700 transition" title="Eliminar">

                                <i class="fa-solid fa-xmark"></i>

                            </button>

                        </div>

                    </div>

                </div>

                `;

            }).join('');

        }

                document.getElementById('theme-icon').className = 'fa-solid fa-sun';

            const html = document.documentElement;

            const isDark = html.classList.toggle('dark');

            localStorage.setItem('execuflow_theme', isDark ? 'dark' : 'light');

            document.getElementById('theme-icon').className = isDark ? 'fa-solid fa-sun' : 'fa-solid fa-moon';

        }

                toast.classList.add('translate-y-20', 'opacity-0');

            }, 3000);

        }
